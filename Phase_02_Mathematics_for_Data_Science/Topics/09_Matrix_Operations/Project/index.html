০৪ সম্পূর্ণ কার্যকর কোড (Full Script)

নিচের কোডটি এক ট্যাপে কপি করে অন্য যেকোনো জায়গায় টেস্ট বা স্টাডি করতে পারেন:

complete_solution.py 
# Matrix Operations: Transpose & Linear Transformations

# 1. ধাপ ১: ম্যাট্রিক্স ট্রান্সপোজ (Transpose)
import numpy as np

A = np.array([
    [1, 2, 3],
    [4, 5, 6]
])

A_T = A.T
print("Original Matrix (2x3):\n", A)
print("Transposed Matrix (3x2):\n", A_T)

# 2. ধাপ ২: স্কেলার গুণন দিয়ে বার্ষিক রূপান্তর
daily_returns = np.array([
    [0.001, 0.0015],
    [0.0008, 0.0012]
])

# ২৫২ ট্রেডিং দিন দিয়ে স্কেলার গুণন
annualized = daily_returns * 252
print("Annualized Return Matrix (%):\n", np.round(annualized * 100, 2))

# 3. ধাপ ৩: ম্যাট্রিক্স সেন্টারিং (De-meaning)
means = np.mean(daily_returns, axis=0)
centered = daily_returns - means

print("Column Means:", means)
print("Centered Matrix (Mean=0):\n", np.round(centered, 6))
