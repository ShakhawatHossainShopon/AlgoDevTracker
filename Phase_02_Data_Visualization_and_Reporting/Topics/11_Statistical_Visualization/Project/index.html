টপিক ১১: স্ট্যাটিস্টিক্যাল কিউ-কিউ (QQ) প্লট ও নরমালিটি টেস্ট — প্র্যাকটিক্যাল ল্যাব

রিটার্নস ডেটা নরমাল ডিস্ট্রিবিউশন মেনে চলে কিনা তা Quantile-Quantile (QQ) প্লটের মাধ্যমে ভিজ্যুয়ালি পরীক্ষা করা।

প্রজেক্ট অবজেক্টিভ: নিচের কোড ব্লকগুলো ধাপে ধাপে পর্যবেক্ষণ করে অ্যালগরিদমটি কীভাবে কাজ করে তা আয়ত্ত করুন। প্রতিটি কোডের জন্য ওয়ান-ট্যাপ কপি বাটন রয়েছে।

০১ scipy.stats probplot দিয়ে QQ প্লট

step_01.py 
import matplotlib.pyplot as plt
import numpy as np
from scipy import stats

np.random.seed(55)
# নন-নরমাল ফিন্যান্সিয়াল রিটার্ন
heavy_tail_returns = np.random.standard_t(df=3, size=250)

fig, ax = plt.subplots(figsize=(6, 5))
res = stats.probplot(heavy_tail_returns, dist="norm", plot=ax)

ax.get_lines()[0].set_markerfacecolor("#2563eb")
ax.get_lines()[0].set_markersize(5)
ax.get_lines()[1].set_color("#dc2626")
ax.get_lines()[1].set_linewidth(2)

ax.set_title("Normal Q-Q Plot of Asset Returns", fontweight="bold")
ax.grid(True, alpha=0.3)

print("QQ Plot Rendered: Curvature at Extremes Indicates Fat Tails.")
Terminal Output
QQ Plot Rendered: Curvature at Extremes Indicates Fat Tails.

complete_pipeline.py 
# 1. scipy.stats probplot দিয়ে QQ প্লট
import matplotlib.pyplot as plt
import numpy as np
from scipy import stats

np.random.seed(55)
# নন-নরমাল ফিন্যান্সিয়াল রিটার্ন
heavy_tail_returns = np.random.standard_t(df=3, size=250)

fig, ax = plt.subplots(figsize=(6, 5))
res = stats.probplot(heavy_tail_returns, dist="norm", plot=ax)

ax.get_lines()[0].set_markerfacecolor("#2563eb")
ax.get_lines()[0].set_markersize(5)
ax.get_lines()[1].set_color("#dc2626")
ax.get_lines()[1].set_linewidth(2)

ax.set_title("Normal Q-Q Plot of Asset Returns", fontweight="bold")
ax.grid(True, alpha=0.3)

print("QQ Plot Rendered: Curvature at Extremes Indicates Fat Tails.")

# 2. হিস্টোগ্রাম বনাম কিউ-কিউ ডায়াগনস্টিক
skew = stats.skew(heavy_tail_returns)
kurt = stats.kurtosis(heavy_tail_returns)

print(f"Empirical Skewness: {skew:.3f}")
print(f"Empirical Excess Kurtosis: {kurt:.3f} (Significant Fat-Tails)")
